  🔬 Proactive Scanner + Pattern Miner - Implementation Design

  🎯 Core Challenge

  The Problem: How do you teach AI what's "interesting" in options data when:
  - You don't know all the patterns yourself
  - Patterns evolve (what worked in 2024 may not work in 2025)
  - False signals are expensive (both $ and attention)
  - You need trustworthy, verifiable output

  The Solution: Multi-layered approach combining:
  1. Rule-based foundations (high confidence, low discovery)
  2. AI hypothesis generation (low confidence, high discovery)
  3. Historical validation (proof required before alerting)
  4. Continuous feedback loops (patterns that work get reinforced)

  def generate_pattern_hypotheses():

  REQUIREMENTS:

  3. Combine multiple signals (OI + flow + time + price action)
  4. Be specific about thresholds (e.g., "OI increase >50%", not "OI increases")

  FORMAT (one per line):
  HYPOTHESIS: [description]
  LOGIC: [why this might work]
  SQL_CONCEPT: [how to detect this in the database]

  EXAMPLES:
  HYPOTHESIS: When call OI builds 5+ days before earnings AND IV percentile <30,
  the stock tends to beat earnings and rally
  LOGIC: Smart money knows something, positioning early while vol is cheap
  SQL_CONCEPT: JOIN earnings_calendar + option_symbol_summary + oi_timing,
  filter for OI build dates before earnings, low IV

  HYPOTHESIS: When large flow alerts (>$200k) DON'T increase OI the next day,
  it signals a one-day hedge, not conviction - fade the signal
  LOGIC: Institutions hedging short-term event, not taking directional position
  SQL_CONCEPT: JOIN flow_alerts + option_contracts OI change,
  WHERE flow premium >200k AND next-day OI change <10%

  YOUR TURN - Generate 10 hypotheses:
  """

      # Call Claude to generate hypotheses
      response = claude_api.call(hypothesis_prompt, model='claude-3-5-sonnet-20241022')

      # Parse hypotheses
      hypotheses = parse_hypotheses(response)

      return hypotheses

  def validate_hypothesis(hypothesis):
      """
      Test if a hypothesis actually predicts profitable trades
      """

      # Step 2: AI converts hypothesis to SQL query
      sql_generation_prompt = f"""
  Convert this trading hypothesis into a SQL query that detects the pattern
  in historical data:

  HYPOTHESIS: {hypothesis['description']}
  LOGIC: {hypothesis['logic']}
  SQL CONCEPT: {hypothesis['sql_concept']}

  DATABASE SCHEMA:
  {schema}

  GENERATE SQL that:
  1. Finds instances where this pattern occurred
  2. Includes the symbol, date pattern was detected
  3. Includes stock price at detection
  4. Includes stock price 7 days later (for profit calculation)

  OUTPUT ONLY THE SQL QUERY:
  """

      sql = claude_api.call(sql_generation_prompt, model='claude-3-5-haiku-20241022')

      # Step 3: Execute SQL and analyze results
      try:
          results = execute_sql(sql)

          # Calculate success metrics
          total_instances = len(results)
          profitable = sum(1 for r in results if r['profit_pct'] > 25)  # Your 25% target
          success_rate = profitable / total_instances if total_instances > 0 else 0
          avg_profit = sum(r['profit_pct'] for r in results) / total_instances if total_instances else 0

          # Step 4: AI analyzes if this pattern is reliable
          analysis_prompt = f"""
  Analyze this trading pattern's historical performance:

  HYPOTHESIS: {hypothesis['description']}

  BACKTEST RESULTS:
  - Total instances found: {total_instances}
  - Profitable trades (>25%): {profitable}
  - Success rate: {success_rate:.1%}
  - Average profit: {avg_profit:.1f}%

  RAW DATA (sample):
  {json.dumps(results[:10], indent=2)}

  QUESTIONS:
  1. Is this pattern statistically significant? (need 15+ instances, >60% success rate)
  2. Are there obvious issues? (survivorship bias, lookahead bias, overfitting?)
  3. What conditions make it work best? (market regime, sector, timeframe?)
  4. Should we add this to the daily scanner? YES/NO and why?

  ANALYSIS:
  """

          analysis = claude_api.call(analysis_prompt, model='claude-3-5-sonnet-20241022')

          return {
              'hypothesis': hypothesis,
              'sql': sql,
              'total_instances': total_instances,
              'success_rate': success_rate,
              'avg_profit': avg_profit,
              'analysis': analysis,
              'validated': success_rate > 0.60 and total_instances > 15
          }

      except Exception as e:
          return {
              'hypothesis': hypothesis,
              'error': str(e),
              'validated': False
          }

  def mine_correlation_patterns():
      """
      Find correlations between database fields and future price movements
      Let AI discover what predicts profits
      """

      # Step 1: Extract features for past 6 months
      features_query = """
          WITH symbol_features AS (
              SELECT
                  oc.symbol,
                  oc.trade_date,
                  oc.trade_date + 7 as future_date,

                  -- OI features
                  SUM(CASE WHEN oc.option_type='call' THEN oc.open_interest ELSE 0 END) as total_call_oi,
                  SUM(CASE WHEN oc.option_type='put' THEN oc.open_interest ELSE 0 END) as total_put_oi,
                  AVG(CASE WHEN oc.option_type='call' THEN oc.oi_change_5d END) as avg_call_oi_change,

                  -- Flow features
                  COUNT(DISTINCT fa.id) as alert_count,
                  MAX(fa.significance_score) as max_significance,
                  SUM(fa.premium_value) as total_premium,

                  -- IV features
                  oss.iv_percentile_30d,
                  oss.iv_front_month,
                  oss.put_call_ratio,

                  -- Price at detection
                  hp.close_price as price_at_detection

              FROM option_contracts oc
              LEFT JOIN flow_alerts fa ON oc.symbol = fa.symbol AND oc.trade_date = fa.trade_date
              LEFT JOIN option_symbol_summary oss ON oc.symbol = oss.symbol AND oc.trade_date = oss.trade_date
              LEFT JOIN historical_prices hp ON oc.symbol = hp.symbol AND oc.trade_date = hp.trade_date
              WHERE oc.trade_date BETWEEN date('now', '-180 days') AND date('now', '-7 days')
              GROUP BY oc.symbol, oc.trade_date
          ),

          with_outcomes AS (
              SELECT
                  sf.*,
                  hp_future.close_price as price_7d_later,
                  ((hp_future.close_price / sf.price_at_detection) - 1) * 100 as profit_pct,
                  CASE WHEN ((hp_future.close_price / sf.price_at_detection) - 1) * 100 > 25 THEN 1 ELSE 0 END
  as is_profitable
              FROM symbol_features sf
              LEFT JOIN historical_prices hp_future
                  ON sf.symbol = hp_future.symbol
                  AND sf.future_date = hp_future.trade_date
          )

          SELECT * FROM with_outcomes
          WHERE price_7d_later IS NOT NULL
      """

      data = execute_sql(features_query)

      # Step 2: AI analyzes correlations

  - avg_call_oi_change (OI momentum)
  - alert_count, max_significance, total_premium (flow data)
  - iv_percentile_30d, iv_front_month, put_call_ratio (volatility)

  TARGET: is_profitable (1 if 7-day return >25%, 0 otherwise)

  SAMPLE DATA:
  {json.dumps(data[:20], indent=2)}

  Look for combinations of features that predict is_profitable=1

  For example:
  - "When avg_call_oi_change >50 AND iv_percentile_30d <30 AND max_significance >7,

  Generate top 5 most predictive patterns with:
  1. Feature combination (specific thresholds)
  2. Success rate (profitable instances / total instances)
  3. Sample size (total instances matching criteria)
  4. Why this makes sense (trading logic)

  PATTERNS:
  """

      patterns = claude_api.call(correlation_prompt, model='claude-3-5-sonnet-20241022')

      # Step 3: Validate each pattern
      # (convert to SQL, re-test, confirm success rate)

      return validated_patterns

  def trustworthy_analysis(symbol, pattern_data):
      """
      Get analysis from 3 different AI models
      Only report if they agree
      """

      models = [
          ('claude-3-5-sonnet-20241022', 'anthropic'),
          ('gpt-4o', 'openai'),
          ('gemini-1.5-pro', 'google')
      ]

      analyses = []
      for model, provider in models:
          analysis = get_analysis(symbol, pattern_data, model, provider)
          analyses.append(analysis)

      # Check for consensus
      recommendations = [a['trade_recommendation']['action'] for a in analyses]

      if len(set(recommendations)) == 1:
          # All agree
          return {
              'consensus': True,
              'recommendation': recommendations[0],
              'analyses': analyses,
              'confidence': 'HIGH - All 3 models agree'
          }
      elif recommendations.count('BUY') >= 2:
          # 2/3 agree
          return {
              'consensus': True,
              'recommendation': 'BUY',
              'analyses': analyses,
              'confidence': 'MEDIUM - 2/3 models agree',
              'dissent': [a for a in analyses if a['trade_recommendation']['action'] != 'BUY']
          }
      else:
          # No consensus
          return {
              'consensus': False,
              'recommendation': 'PASS',
              'analyses': analyses,
              'confidence': 'LOW - Models disagree',
              'reason': 'Insufficient AI consensus to recommend trade'
          }

  This ensures:
  - ✅ Multiple perspectives (not one AI's bias)
