-- View: v_option_comparison
-- Purpose: Side-by-side comparison of options for watchlist symbols
-- Created by: morning_views.py (auto-recreated each run)
-- Last Updated: 2025-10-07
-- Refreshed: Every time morning_views.py runs

-- DATA STRATEGY: Hybrid approach
--   - OI fields: FROM TODAY (fresh positioning)
--   - Volume/Price/Greeks: FROM YESTERDAY (complete data)
--
-- Shows detailed contract-level metrics for comparing options within the same symbol.
-- Useful for evaluating different strikes/expirations when choosing a position.


CREATE VIEW v_option_comparison AS
SELECT
    -- IDENTIFICATION --
    o_today.contract_hash,                             -- Unique contract identifier
    o_today.symbol,
    o_today.trade_date,                                -- Date of OI data (today)
    o_today.strike,
    o_today.expiration_date,
    o_today.option_type,                               -- CALL or PUT
    o_today.days_to_expiration AS dte,
    o_yesterday.moneyness,                             -- ITM/ATM/OTM classification

    -- PRICING (from yesterday - complete) --
    o_yesterday.last_price,                            -- Option premium
    o_yesterday.underlying_price,                      -- Stock price

    -- GREEKS (from yesterday - complete) --
    o_yesterday.delta,                                 -- Price sensitivity
    o_yesterday.gamma,                                 -- Delta acceleration
    o_yesterday.theta,                                 -- Time decay
    o_yesterday.vega,                                  -- IV sensitivity
    o_yesterday.implied_volatility AS IV,              -- Current IV level

    -- VOLUME & OI (hybrid) --
    o_today.open_interest,                             -- Current OI (from today)
    o_yesterday.volume,                                -- Yesterday's volume

    -- BREAKEVEN ANALYSIS --
    CASE
        WHEN o_today.option_type = 'CALL' THEN o_today.strike + o_yesterday.last_price
        WHEN o_today.option_type = 'PUT' THEN o_today.strike - o_yesterday.last_price
    END as breakeven_price,
    -- Price stock must reach by expiration to break even

    CASE
        WHEN o_today.option_type = 'CALL' THEN ROUND(((o_today.strike + o_yesterday.last_price) / o_yesterday.underlying_price - 1) * 100, 2)
        WHEN o_today.option_type = 'PUT' THEN ROUND((1 - (o_today.strike - o_yesterday.last_price) / o_yesterday.underlying_price) * 100, 2)
    END as breakeven_move_pct,
    -- Percentage move required to break even

    -- GREEK EFFICIENCY METRICS --
    ROUND(ABS(o_yesterday.delta) / NULLIF(o_yesterday.last_price, 0), 2) as delta_per_dollar,
    -- Delta exposure per $1 spent (higher = more leverage)

    CASE WHEN o_yesterday.gamma > 0.05 THEN 1 ELSE 0 END as in_gamma_zone,
    -- Binary flag: Is this contract in high-gamma zone?

    ROUND(o_yesterday.theta * 100, 2) as theta_decay_dollars,
    -- Daily time decay in dollars (negative = losing money)

    ROUND(o_yesterday.vega * 100, 2) as vega_dollars_per_iv_point,
    -- Dollars gained/lost per 1% move in IV

    -- VALUE DECOMPOSITION --
    CASE
        WHEN o_today.option_type = 'CALL' AND o_yesterday.underlying_price > o_today.strike
        THEN ROUND(o_yesterday.underlying_price - o_today.strike, 2)
        WHEN o_today.option_type = 'PUT' AND o_yesterday.underlying_price < o_today.strike
        THEN ROUND(o_today.strike - o_yesterday.underlying_price, 2)
        ELSE 0
    END as intrinsic_value,
    -- How much contract is worth if exercised immediately

    ROUND(o_yesterday.last_price - CASE
        WHEN o_today.option_type = 'CALL' AND o_yesterday.underlying_price > o_today.strike
        THEN o_yesterday.underlying_price - o_today.strike
        WHEN o_today.option_type = 'PUT' AND o_yesterday.underlying_price < o_today.strike
        THEN o_today.strike - o_yesterday.underlying_price
        ELSE 0
    END, 2) as extrinsic_value,
    -- Time value + IV premium (this is what theta decays)

    -- LIQUIDITY --
    s_yesterday.avg_bid_ask_spread_pct                 -- Average spread (NOTE: Currently NULL)

FROM oi_daily o_today
JOIN oi_daily o_yesterday
    ON o_today.contract_hash = o_yesterday.contract_hash
    AND o_yesterday.trade_date = (SELECT MAX(trade_date) FROM oi_daily WHERE trade_date < DATE('now'))
JOIN oi_symbol_summary s_yesterday
    ON o_yesterday.symbol = s_yesterday.symbol
    AND o_yesterday.trade_date = s_yesterday.trade_date

WHERE
    o_today.days_to_expiration BETWEEN 7 AND 60        -- FILTER: 1-8 week options (sweet spot)
    AND o_yesterday.last_price > 0                     -- FILTER: Must have valid pricing
    AND o_today.trade_date = DATE('now');

-- INTERPRETATION GUIDE:
--
-- Example: MGM $36 CALL (10/24)
--   Last Price: $0.50
--   Underlying: $33.93
--   Delta: 0.32
--   Breakeven: $36.50 (+7.6% move required)
--   Delta per Dollar: 0.64 (32 delta for $50 cost)
--   Theta: -$0.04/day
--   Intrinsic: $0.00 (OTM)
--   Extrinsic: $0.50 (100% time value)
--
-- Interpretation:
-- - Need 7.6% move to break even by 10/24 (10 days out)
-- - Every $1 move in MGM = $0.32 gain (delta)
-- - Losing $4/day to time decay (theta)
-- - High delta per dollar (0.64) = efficient leverage
-- - 100% extrinsic = pure speculation on direction + IV


-- KEY METRICS EXPLAINED:
--
-- 1. BREAKEVEN MOVE PCT
--    - Percentage stock must move by expiration to profit
--    - Lower is better (easier to achieve)
--    - Compare to historical volatility and expected moves
--
-- 2. DELTA PER DOLLAR
--    - Directional exposure per $1 invested
--    - Higher = more leverage (riskier but more potential)
--    - Compare strikes: deeper ITM = lower delta/$ but safer
--
-- 3. INTRINSIC vs EXTRINSIC
--    - Intrinsic = guaranteed value if stock stays here
--    - Extrinsic = what you're paying for time and volatility
--    - High extrinsic = expensive, needs move or IV expansion
--
-- 4. THETA DECAY
--    - How much you lose per day if nothing changes
--    - Critical for short-term trades (< 21 DTE)
--    - Accelerates as expiration approaches
--
-- 5. GAMMA ZONE
--    - High gamma = delta changes rapidly near this strike
--    - Good: Fast gains if directional move happens
--    - Bad: Fast losses if wrong direction
--    - Typically ATM contracts in 7-21 DTE range
