-- View: v_symbol_oi_detail
-- Purpose: Detailed OI distribution pattern for a specific symbol
-- Created by: morning_views.py (auto-recreated each run)
-- Last Updated: 2025-10-07
-- Refreshed: Every time morning_views.py runs

-- DATA STRATEGY: Hybrid approach
--   - OI fields: FROM TODAY (fresh positioning)
--   - Volume/Price/Greeks: FROM YESTERDAY (complete data)
--
-- Shows the complete OI pattern across the entire options chain for one symbol.
-- Used when doing deep-dive analysis on a specific symbol from watchlist.


CREATE VIEW v_symbol_oi_detail AS
SELECT
    -- IDENTIFICATION --
    s_today.symbol,
    s_yesterday.trade_date,                            -- Date of complete data
    s_yesterday.close_price,                           -- Stock price

    -- OPEN INTEREST TOTALS (from today - fresh) --
    s_today.total_open_interest,
    s_today.total_call_oi,
    s_today.total_put_oi,
    s_today.put_call_ratio,
    s_today.oi_balance_text,                           -- Text description of OI balance

    -- TOP STRIKES (from today - fresh) --
    s_today.top_call_strike,                           -- Highest call OI strike
    s_today.top_call_expiration,                       -- Expiration of top call
    s_today.top_call_oi,                               -- OI at top call strike
    s_today.top_call_pct_of_total,                     -- % of total OI at top call
    s_today.top_call_display,                          -- Formatted: "$28.0 (10/17) OI 9,619 | $29.0 (10/17) OI 24,239 | ..."

    s_today.top_put_strike,                            -- Highest put OI strike
    s_today.top_put_expiration,                        -- Expiration of top put
    s_today.top_put_oi,                                -- OI at top put strike
    s_today.top_put_pct_of_total,                      -- % of total OI at top put
    s_today.top_put_display,                           -- Formatted: "$24.0 (10/17) OI 2,315 | $27.0 (10/17) OI 3,955 | ..."

    -- TIME DISTRIBUTION (from today - fresh) --
    -- Shows when contracts expire (DTE buckets)
    s_today.oi_0_7_days,                               -- OI expiring 0-7 days (absolute)
    s_today.oi_8_21_days,                              -- OI expiring 8-21 days (absolute)
    s_today.oi_22_35_days,                             -- OI expiring 22-35 days (absolute)
    s_today.oi_36_60_days,                             -- OI expiring 36-60 days (absolute)
    s_today.oi_0_7_days_percent,                       -- % in 0-7 DTE
    s_today.oi_8_21_days_percent,                      -- % in 8-21 DTE
    s_today.oi_22_35_days_percent,                     -- % in 22-35 DTE
    s_today.oi_36_60_days_percent,                     -- % in 36-60 DTE

    -- TIME DISTRIBUTION BY SIDE (from today - fresh) --
    s_today.call_oi_0_7_days,
    s_today.call_oi_8_21_days,
    s_today.call_oi_22_35_days,
    s_today.call_oi_36_60_days,
    s_today.put_oi_0_7_days,
    s_today.put_oi_8_21_days,
    s_today.put_oi_22_35_days,
    s_today.put_oi_36_60_days,

    -- VOLUME (from yesterday - complete) --
    -- NOTE: Currently all 0 (needs aggregation fix)
    s_yesterday.option_volume,                         -- Total option volume
    s_yesterday.call_volume,                           -- Call volume
    s_yesterday.put_volume,                            -- Put volume
    s_yesterday.volume_put_call_ratio,                 -- Volume-based put/call ratio

    -- IMPLIED VOLATILITY (from yesterday - complete) --
    s_yesterday.call_iv_avg,                           -- Average call IV
    s_yesterday.put_iv_avg,                            -- Average put IV
    s_yesterday.iv_skew,                               -- Put IV - Call IV
    s_yesterday.symbol_iv_percentile_30d,              -- Where current IV ranks (30d lookback)

    -- GREEK EXPOSURES (from yesterday - complete) --
    s_yesterday.total_delta_exposure,                  -- Total delta exposure (shares)
    s_yesterday.call_delta_exposure,                   -- Call delta exposure
    s_yesterday.put_delta_exposure,                    -- Put delta exposure
    s_yesterday.net_delta_exposure,                    -- Net delta (call - put)
    s_yesterday.total_gamma_exposure,                  -- Total gamma exposure
    s_yesterday.max_gamma_strike,                      -- Strike with most gamma concentration
    s_yesterday.total_theta_exposure,                  -- Total theta decay
    s_yesterday.total_vega_exposure,                   -- Total vega exposure
    s_yesterday.call_vega_exposure,                    -- Call vega
    s_yesterday.put_vega_exposure,                     -- Put vega
    s_yesterday.net_vega_exposure,                     -- Net vega

    -- MONEYNESS DISTRIBUTION - ABSOLUTE (from today - fresh) --
    s_today.deep_itm_call_oi,                          -- Deep ITM call OI (absolute)
    s_today.itm_call_oi,                               -- ITM call OI
    s_today.atm_call_oi,                               -- ATM call OI
    s_today.otm_call_oi,                               -- OTM call OI
    s_today.deep_otm_call_oi,                          -- Deep OTM call OI

    s_today.deep_itm_put_oi,                           -- Deep ITM put OI (absolute)
    s_today.itm_put_oi,                                -- ITM put OI
    s_today.atm_put_oi,                                -- ATM put OI
    s_today.otm_put_oi,                                -- OTM put OI
    s_today.deep_otm_put_oi,                           -- Deep OTM put OI

    -- MONEYNESS DISTRIBUTION - PERCENTAGES (from today - fresh) --
    s_today.deep_itm_call_pct,                         -- % of call OI that is deep ITM
    s_today.itm_call_pct,                              -- % ITM
    s_today.atm_call_pct,                              -- % ATM
    s_today.otm_call_pct,                              -- % OTM
    s_today.deep_otm_call_pct,                         -- % deep OTM

    s_today.deep_itm_put_pct,                          -- % of put OI that is deep ITM
    s_today.itm_put_pct,                               -- % ITM
    s_today.atm_put_pct,                               -- % ATM
    s_today.otm_put_pct,                               -- % OTM
    s_today.deep_otm_put_pct,                          -- % deep OTM

    -- LIQUIDITY & MAX PAIN (mixed sources) --
    s_yesterday.avg_bid_ask_spread_pct,                -- Average bid-ask spread (NOTE: Currently NULL)
    s_today.max_pain_by_friday                         -- Max pain level (where most OI expires worthless)

FROM oi_symbol_summary s_today
JOIN oi_symbol_summary s_yesterday
    ON s_today.symbol = s_yesterday.symbol
    AND s_yesterday.trade_date = (SELECT MAX(trade_date) FROM oi_symbol_summary WHERE trade_date < DATE('now'))
WHERE s_today.trade_date = DATE('now');


-- USAGE EXAMPLES:
-- 1. Get detail for specific symbol:
--    SELECT * FROM v_symbol_oi_detail WHERE symbol = 'KDP';
--
-- 2. Get OI distribution for symbol:
--    SELECT symbol, oi_0_7_days_percent, oi_8_21_days_percent, oi_22_35_days_percent
--    FROM v_symbol_oi_detail WHERE symbol = 'MGM';
--
-- 3. Check Greek exposures:
--    SELECT symbol, net_delta_exposure, total_gamma_exposure, max_gamma_strike
--    FROM v_symbol_oi_detail WHERE symbol = 'AAL';


-- INTERPRETATION NOTES:
-- 1. Time Distribution: Where is OI concentrated by expiration?
--    - High % in 0-7 DTE = Near-term event expected
--    - High % in 36-60 DTE = Long-term positioning
--
-- 2. Moneyness: Are positions ITM, ATM, or OTM?
--    - High OTM call % = Speculative bullish bets
--    - High ITM put % = Hedging or bearish conviction
--
-- 3. Max Pain: Where will stock gravitate to expire most options worthless?
--    - Compare to current price for potential move direction
--
-- 4. Greeks: What's the net exposure?
--    - Positive net delta = Net bullish positioning
--    - High gamma = Rapid delta changes near this price
--    - Positive net vega = Benefits from IV expansion

-- Tables:
--   - oi_symbol_summary (both today and yesterday)
