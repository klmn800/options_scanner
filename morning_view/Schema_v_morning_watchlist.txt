-- View: v_morning_watchlist
-- Purpose: Daily ranked watchlist of symbols worth manual analysis
-- Created by: morning_views.py (auto-recreated each run)
-- Last Updated: 2025-10-07
-- Refreshed: Every time morning_views.py runs (drops and recreates view)

-- DATA STRATEGY: Hybrid approach for most accurate morning picture
--   - OI fields: FROM TODAY (s_today.trade_date = DATE('now'))
--   - Volume/Price/Greeks: FROM YESTERDAY (s_yesterday.trade_date < DATE('now'))
--
-- Why? Morning OID scan updates OI only. Volume and greeks populate in evening.
-- Using yesterday's complete data + today's fresh OI gives best decision context.

-- SELF-JOIN APPROACH:
--   s_today = oi_symbol_summary WHERE trade_date = DATE('now')
--   s_yesterday = oi_symbol_summary WHERE trade_date = MAX(trade_date) < DATE('now')
--   Joined on symbol match


CREATE VIEW v_morning_watchlist AS
SELECT
    -- IDENTIFICATION --
    s_today.symbol,                                    -- Stock symbol

    -- PRICING (from yesterday - complete data) --
    s_yesterday.close_price,                           -- Last close price
    s_yesterday.option_volume as volume,               -- Total option volume (NOTE: Currently 0, needs fix)

    (SELECT ROUND(((s_yesterday.close_price - h5.close_price) / NULLIF(h5.close_price, 0)) * 100, 1)
     FROM historical_prices h5
     WHERE h5.symbol = s_yesterday.symbol
     AND h5.trade_date < s_yesterday.trade_date
     ORDER BY h5.trade_date DESC
     LIMIT 1 OFFSET 4
    ) as price_change_5d_pct,                          -- 5-day price change percentage

    -- OPEN INTEREST (from today - fresh positioning) --
    s_today.put_call_ratio,                            -- Put/Call ratio by OI
    s_today.total_open_interest,                       -- Total contracts open
    s_today.total_call_oi,                             -- Call contracts open
    s_today.total_put_oi,                              -- Put contracts open

    -- DIRECTIONAL BIAS (calculated from today's OI) --
    CASE
        WHEN s_today.put_call_ratio > 2.0 THEN 'BEARISH'
        WHEN s_today.put_call_ratio > 1.3 THEN 'SOMEWHAT_BEARISH'
        WHEN s_today.put_call_ratio < 0.5 THEN 'BULLISH'
        WHEN s_today.put_call_ratio < 0.8 THEN 'SOMEWHAT_BULLISH'
        ELSE 'NEUTRAL'
    END as direction_bias,

    -- CONVICTION LEVEL (based on OI concentration) --
    CASE
        WHEN s_today.top_call_pct_of_total > 30 OR s_today.top_put_pct_of_total > 30 THEN 'HIGH'
        WHEN s_today.top_call_pct_of_total > 15 OR s_today.top_put_pct_of_total > 15 THEN 'MEDIUM'
        ELSE 'LOW'
    END as conviction_level,

    -- EARNINGS CATALYST --
    e.earnings_days_ahead,                             -- Days until earnings (NULL if none)
    e.straddle_expected_move_pct AS exp_move_pct,     -- Expected move from IV
    e.historical_avg_move_pct AS hist_move_pct,       -- Historical average move
    e.move_difference_pct AS diff_move_pct,           -- Difference (expected - historical)
    e.earnings_play_signal,                            -- PLAY/AVOID signal

    -- FLOW ALERTS (last 7 days) --
    (SELECT COUNT(*) FROM flow_alerts fa
     WHERE fa.symbol = s_today.symbol
     AND fa.evaluation_status = 'active'
     AND DATE(fa.alert_timestamp) >= DATE('now', '-7 days')
    ) as active_alerts_count,

    -- NEWS SENTIMENT (last 5 days) --
    (SELECT AVG(symbol_sentiment_score)
     FROM news_symbol_sentiment ns
     WHERE ns.symbol = s_today.symbol
     AND DATE(ns.article_date) >= DATE('now', '-5 days')
    ) as news_sentiment_avg,

    CASE
        WHEN (SELECT AVG(symbol_sentiment_score)
              FROM news_symbol_sentiment ns
              WHERE ns.symbol = s_today.symbol
              AND DATE(ns.article_date) >= DATE('now', '-5 days')) >= 0.15 THEN 'Bullish'
        WHEN (SELECT AVG(symbol_sentiment_score)
              FROM news_symbol_sentiment ns
              WHERE ns.symbol = s_today.symbol
              AND DATE(ns.article_date) >= DATE('now', '-5 days')) >= 0.05 THEN 'Somewhat-Bullish'
        WHEN (SELECT AVG(symbol_sentiment_score)
              FROM news_symbol_sentiment ns
              WHERE ns.symbol = s_today.symbol
              AND DATE(ns.article_date) >= DATE('now', '-5 days')) <= -0.15 THEN 'Bearish'
        WHEN (SELECT AVG(symbol_sentiment_score)
              FROM news_symbol_sentiment ns
              WHERE ns.symbol = s_today.symbol
              AND DATE(ns.article_date) >= DATE('now', '-5 days')) <= -0.05 THEN 'Somewhat-Bearish'
        ELSE 'Neutral'
    END as news_sentiment,

    -- VOLUME SURGE (from yesterday's complete data) --
    -- NOTE: Currently returns NULL because option_volume = 0 (needs fix)
    ROUND(s_yesterday.option_volume / NULLIF((SELECT AVG(s2.option_volume)
                                    FROM oi_symbol_summary s2
                                    WHERE s2.symbol = s_yesterday.symbol
                                    AND s2.trade_date BETWEEN DATE(s_yesterday.trade_date, '-20 days') AND DATE(s_yesterday.trade_date, '-1 days')), 0), 2) as volume_surge_factor,

    -- CONFLUENCE SCORE (0-5 points) --
    -- Counts how many signals are aligned (more = higher conviction)
    (
        CASE WHEN (SELECT COUNT(*) FROM flow_alerts fa
                   WHERE fa.symbol = s_today.symbol
                   AND fa.evaluation_status = 'active'
                   AND DATE(fa.alert_timestamp) >= DATE('now', '-7 days')) > 0
        THEN 1 ELSE 0 END                              -- +1 for flow alerts
        +
        CASE WHEN s_yesterday.option_volume > (SELECT AVG(s2.option_volume) * 1.5
                                     FROM oi_symbol_summary s2
                                     WHERE s2.symbol = s_yesterday.symbol
                                     AND s2.trade_date BETWEEN DATE(s_yesterday.trade_date, '-20 days') AND DATE(s_yesterday.trade_date, '-1 days'))
        THEN 1 ELSE 0 END                              -- +1 for volume surge (NOTE: currently never fires due to 0 volume)
        +
        CASE WHEN e.earnings_days_ahead BETWEEN 1 AND 30 THEN 1 ELSE 0 END  -- +1 for earnings catalyst
        +
        CASE WHEN ABS((SELECT AVG(symbol_sentiment_score)
                       FROM news_symbol_sentiment ns
                       WHERE ns.symbol = s_today.symbol
                       AND DATE(ns.article_date) >= DATE('now', '-5 days'))) >= 0.10
        THEN 1 ELSE 0 END                              -- +1 for strong sentiment (bullish OR bearish)
        +
        CASE WHEN s_today.top_call_pct_of_total > 30 OR s_today.top_put_pct_of_total > 30 THEN 1 ELSE 0 END  -- +1 for high OI conviction
    ) as confluence_score,

    -- PRIMARY SIGNAL (what flagged this symbol) --
    CASE
        WHEN (SELECT COUNT(*) FROM flow_alerts fa
              WHERE fa.symbol = s_today.symbol
              AND fa.evaluation_status = 'active'
              AND DATE(fa.alert_timestamp) >= DATE('now', '-7 days')) > 0
        THEN 'FLOW_ALERT'
        WHEN e.earnings_days_ahead BETWEEN 1 AND 30 THEN 'EARNINGS_CATALYST'
        WHEN s_yesterday.option_volume > (SELECT AVG(s2.option_volume) * 1.5
                                FROM oi_symbol_summary s2
                                WHERE s2.symbol = s_yesterday.symbol
                                AND s2.trade_date BETWEEN DATE(s_yesterday.trade_date, '-20 days') AND DATE(s_yesterday.trade_date, '-1 days'))
        THEN 'VOLUME_SURGE'                            -- NOTE: Never fires due to 0 volume
        ELSE 'OI_SIGNAL'
    END as primary_signal,

    -- MARKET CONTEXT --
    COALESCE(m.market_direction, 'NEUTRAL') as market_direction,        -- Bull/Neutral/Bear
    COALESCE(m.regime_classification, 'normal') as market_regime,       -- high_vol/low_vol/normal

    -- SYMBOL CHARACTERISTICS --
    meta.sector,                                       -- Sector classification
    meta.industry,                                     -- Industry classification

    s_yesterday.trade_date                             -- Date of displayed data

FROM oi_symbol_summary s_today
JOIN oi_symbol_summary s_yesterday
    ON s_today.symbol = s_yesterday.symbol
    AND s_yesterday.trade_date = (SELECT MAX(trade_date) FROM oi_symbol_summary WHERE trade_date < DATE('now'))
JOIN symbol_metadata meta ON s_today.symbol = meta.symbol
LEFT JOIN earnings_upcoming e ON s_today.symbol = e.symbol
LEFT JOIN market_daily_summary m ON DATE(m.trade_date) = DATE(s_yesterday.trade_date)

WHERE
    s_today.trade_date = DATE('now')                   -- Today's OI data
    AND s_yesterday.close_price < 60                   -- FILTER: Budget constraint (configurable)
    AND s_today.total_open_interest > 500             -- FILTER: Liquidity requirement (configurable)
    AND meta.is_etf = 0                                -- FILTER: Exclude ETFs

ORDER BY confluence_score DESC, active_alerts_count DESC  -- SORT: Best signals first
LIMIT 20;                                              -- Top 20 symbols only
