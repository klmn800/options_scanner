-- CRITICAL FEATURE: Per Ben's feedback, this is the "killer feature" that justifies the entire system.
--
-- Answers the key question: "Did people buy $12 puts when stock was $13.50 (predictive)
-- or when stock was $11.30 (chasing)?"
--
-- DATA STRATEGY: Hybrid approach
--   - OI and timing: FROM TODAY (o_today.oi_build_start_date populated by timing calculator)
--   - Current price: FROM YESTERDAY (complete pricing data)
--
-- FILTERS: Only shows contracts from watchlist symbols with OI > 1000


CREATE VIEW v_oi_timing_context AS
SELECT
    -- IDENTIFICATION --
    o_today.contract_hash,                             -- Unique contract identifier
    o_today.symbol,
    o_today.strike,
    o_today.option_type,                               -- CALL or PUT
    o_today.expiration_date,
    o_today.trade_date,
    o_today.open_interest,                             -- Current OI level
    o_today.days_to_expiration as dte,

    -- OI TIMING (from today - calculated by oid_timing_calculator.py) --
    o_today.oi_build_start_date,                       -- When OI first hit 50% of current level
    o_today.oi_build_start_price,                      -- Stock price when OI building started

    -- CURRENT CONTEXT (from yesterday - complete data) --
    o_yesterday.underlying_price as current_price,     -- Stock price now

    -- ANALYSIS: PREDICTIVE vs CHASING --
    -- Logic: Compare build price to current price
    CASE
        -- PUT Analysis:
        WHEN o_today.option_type = 'PUT' AND o_today.oi_build_start_price > o_yesterday.underlying_price
        THEN 'PREDICTIVE (bought puts when stock was higher)'
        -- Bought puts when stock was higher, now stock fell = predicted the move

        WHEN o_today.option_type = 'PUT' AND o_today.oi_build_start_price < o_yesterday.underlying_price
        THEN 'CHASING (bought puts after stock fell)'
        -- Bought puts after stock already fell = chasing the move (possibly bagholders)

        -- CALL Analysis:
        WHEN o_today.option_type = 'CALL' AND o_today.oi_build_start_price < o_yesterday.underlying_price
        THEN 'PREDICTIVE (bought calls when stock was lower)'
        -- Bought calls when stock was lower, now stock rose = predicted the move

        WHEN o_today.option_type = 'CALL' AND o_today.oi_build_start_price > o_yesterday.underlying_price
        THEN 'CHASING (bought calls after stock rose)'
        -- Bought calls after stock rose = momentum/FOMO play (possibly bagholders if peaked)

        ELSE 'NEUTRAL'
    END as positioning_type,

    -- TIME ELAPSED --
    CAST(JULIANDAY(o_today.trade_date) - JULIANDAY(o_today.oi_build_start_date) AS INTEGER) as oi_build_days_since,
    -- How many days ago was this position established?

    -- PRICE MOVEMENT SINCE BUILD --
    ROUND((o_yesterday.underlying_price - o_today.oi_build_start_price) / o_today.oi_build_start_price * 100, 2) as oi_build_price_move_pct
    -- % move from build price to current price
    -- Positive = stock up since build
    -- Negative = stock down since build

FROM oi_daily o_today
JOIN oi_daily o_yesterday
    ON o_today.contract_hash = o_yesterday.contract_hash
    AND o_yesterday.trade_date = (SELECT MAX(trade_date) FROM oi_daily WHERE trade_date < DATE('now'))

WHERE
    o_today.symbol IN (SELECT symbol FROM v_morning_watchlist)  -- FILTER: Only watchlist symbols
    AND o_today.open_interest > 1000                            -- FILTER: Significant positions only
    AND o_today.oi_build_start_date IS NOT NULL                 -- FILTER: Must have timing data
    AND o_today.trade_date = DATE('now');

-- INTERPRETATION GUIDE:
--
-- Example 1: MGM $36 CALL (10/24)
--   Build date: 2025-10-03 (3 days ago)
--   Build price: $34.00
--   Current price: $33.93
--   Type: CHASING (bought calls after stock rose)
--   Move since: -0.2%
--
-- Interpretation: Fresh positioning (3 days), bought near current level, looking for continuation.
-- Not underwater yet, conviction play expecting more upside.
--
-- Example 2: AAL $12 PUT (10/31)
--   Build date: 2025-08-15 (48 days ago)
--   Build price: $13.50
--   Current price: $11.27
--   Type: PREDICTIVE (bought puts when stock was higher)
--   Move since: -16.5%
--
-- Interpretation: Smart money positioned 48 days ago, predicted the -16.5% drop.
-- Position is deep in profit, likely holding for more downside or taking profits.
--
-- Example 3: AAL $12 PUT (10/31)
--   Build date: 2025-10-01 (2 days ago)
--   Build price: $11.30
--   Current price: $11.27
--   Type: CHASING (bought puts after stock fell)
--   Move since: -0.3%
--
-- Interpretation: Recent retail chase, bought after stock already fell from $13.50.
-- Late to the party, minimal movement since entry. Possible bagholders if bounce.


-- CRITICAL INSIGHTS:
-- 1. PREDICTIVE + Long time ago + Large move = Smart money was right
-- 2. PREDICTIVE + Recent + Small move = Early positioning, watching for catalyst
-- 3. CHASING + Recent + Small move = Momentum play, looking for continuation
-- 4. CHASING + Long time ago + Negative move = Bagholders, stuck in losing position
--
-- Fresh builds (0-7 days) are more actionable than old builds (25+ days)


-- KNOWN LIMITATIONS:
-- 1. "CHASING" label doesn't account for post-purchase behavior
--    - Could be chasing a peak (bad) or riding momentum (good)
--    - Use oi_build_price_move_pct to determine if underwater or profitable
--
-- 2. 50% threshold is arbitrary
--    - Captures substantial builds, but misses early accumulation
--    - Good enough for distinguishing smart money from retail

-- 3. Timing calculator takes time to run
--    - Only runs for contracts with OI > 1000
